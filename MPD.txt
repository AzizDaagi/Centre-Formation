CREATE TABLE FORMATEUR (
    ID_FORMATEUR NUMBER GENERATED BY DEFAULT AS IDENTITY,
    NOM VARCHAR2(100) NOT NULL,
    PRENOM VARCHAR2(100) NOT NULL,
    EMAIL VARCHAR2(255) NOT NULL,
    PASSWORD_HASH VARCHAR2(255) NOT NULL,
    ROLE VARCHAR2(30) DEFAULT 'FORMATEUR' NOT NULL,
    STATUT_COMPTE VARCHAR2(20) DEFAULT 'ACTIF' NOT NULL,

    CONSTRAINT PK_FORMATEUR PRIMARY KEY (ID_FORMATEUR),
    CONSTRAINT UQ_FORMATEUR_EMAIL UNIQUE (EMAIL),
    CONSTRAINT CK_FORMATEUR_ROLE
        CHECK (ROLE IN ('FORMATEUR', 'ADMIN', 'SUPER_ADMIN')),
    CONSTRAINT CK_FORMATEUR_STATUT
        CHECK (STATUT_COMPTE IN ('ACTIF', 'INACTIF'))
);

CREATE TABLE SALLE (
    ID_SALLE NUMBER GENERATED BY DEFAULT AS IDENTITY,
    NOM_SALLE VARCHAR2(150) NOT NULL,
    CAPACITE NUMBER NOT NULL,
    TYPE_SALLE VARCHAR2(50) NOT NULL,
    STATUT VARCHAR2(30) DEFAULT 'DISPONIBLE' NOT NULL,
    REPORT_STATUS VARCHAR2(30) DEFAULT 'NONE',
    REPORT_DESCRIPTION VARCHAR2(1000),
    REPORT_AUTHOR VARCHAR2(255),

    CONSTRAINT PK_SALLE PRIMARY KEY (ID_SALLE),
    CONSTRAINT CK_SALLE_CAPACITE CHECK (CAPACITE > 0)
);

CREATE TABLE COURS (
    ID_COURS NUMBER GENERATED BY DEFAULT AS IDENTITY,
    TITRE VARCHAR2(200) NOT NULL,
    DESCRIPTION VARCHAR2(2000),
    HEURES_REQUISES NUMBER NOT NULL,
    REPORT_STATUS VARCHAR2(30) DEFAULT 'NONE',
    REPORT_DESCRIPTION VARCHAR2(1000),
    REPORT_AUTHOR VARCHAR2(255),
    ID_FORMATEUR_RESP NUMBER,

    CONSTRAINT PK_COURS PRIMARY KEY (ID_COURS),
    CONSTRAINT FK_COURS_FORMATEUR
        FOREIGN KEY (ID_FORMATEUR_RESP)
        REFERENCES FORMATEUR(ID_FORMATEUR)
        ON DELETE SET NULL,
    CONSTRAINT CK_COURS_HEURES CHECK (HEURES_REQUISES > 0)
);

CREATE TABLE STAGIAIRE (
    ID_STAGIAIRE NUMBER GENERATED BY DEFAULT AS IDENTITY,
    NOM VARCHAR2(100) NOT NULL,
    PRENOM VARCHAR2(100) NOT NULL,
    EMAIL VARCHAR2(255) NOT NULL,
    PASSWORD_HASH VARCHAR2(255) NOT NULL,
    DATE_DEBUT DATE,
    DATE_FIN_PREVUE DATE,
    HEURES_VALIDEES NUMBER DEFAULT 0 NOT NULL,
    STATUT VARCHAR2(30) DEFAULT 'ACTIF' NOT NULL,
    ID_FORMATEUR NUMBER,
    ID_COURS NUMBER,
    ID_SALLE_ATTITREE NUMBER,

    CONSTRAINT PK_STAGIAIRE PRIMARY KEY (ID_STAGIAIRE),
    CONSTRAINT UQ_STAGIAIRE_EMAIL UNIQUE (EMAIL),
    CONSTRAINT FK_STAGIAIRE_FORMATEUR
        FOREIGN KEY (ID_FORMATEUR)
        REFERENCES FORMATEUR(ID_FORMATEUR)
        ON DELETE SET NULL,
    CONSTRAINT FK_STAGIAIRE_COURS
        FOREIGN KEY (ID_COURS)
        REFERENCES COURS(ID_COURS)
        ON DELETE SET NULL,
    CONSTRAINT FK_STAGIAIRE_SALLE
        FOREIGN KEY (ID_SALLE_ATTITREE)
        REFERENCES SALLE(ID_SALLE)
        ON DELETE SET NULL,
    CONSTRAINT CK_STAGIAIRE_HEURES CHECK (HEURES_VALIDEES >= 0)
);
